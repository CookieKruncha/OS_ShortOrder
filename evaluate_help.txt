usage: python -m kitchen.cli evaluate [-h] [--seeds SEEDS] [--workers WORKERS]
                                      [--jsonl JSONL] [--json]
                                      [--config CONFIG] [--cores CORES]
                                      [--switch-cost SWITCH_COST]
                                      [--ticks TICKS] [--blind]
                                      scheduler

positional arguments:
  scheduler

options:
  -h, --help            show this help message and exit
  --seeds SEEDS         range START..END
  --workers WORKERS
  --jsonl JSONL         write result rows to this file
  --json
  --config CONFIG       TOML profile (default: the course profile)
  --cores CORES         override the number of cooks
  --switch-cost SWITCH_COST
                        override the context-switch cost
  --ticks TICKS         override the length of the run
  --blind               hide step durations from the scheduler
